import java.util.Scanner;

// BankAccount class
class BankAccount {
    private double balance;

    public BankAccount(double initialBalance) {
        if (initialBalance >= 0) {
            balance = initialBalance;
        } else {
            balance = 0;
        }
    }

    // Deposit money
    public void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            System.out.printf("Deposit successful! Amount: Rs. %.2f%n", amount);
        } else {
            System.out.println("Invalid amount! Enter a positive amount.");
        }
    }

    // Withdraw money
    public void withdraw(double amount) {
        if (amount <= 0) {
            System.out.println("Invalid amount! Enter a positive amount.");
        } else if (amount > balance) {
            System.out.println("Insufficient balance!");
        } else {
            balance -= amount;
            System.out.printf("Withdrawal successful! Amount: Rs. %.2f%n", amount);
        }
    }

    // Check balance
    public void checkBalance() {
        System.out.printf("Your current balance is: Rs. %.2f%n", balance);
    }
}

// ATM class
class ATM {
    private final BankAccount account;
    private final Scanner scanner;

    public ATM(BankAccount account, Scanner scanner) {
        this.account = account;
        this.scanner = scanner;
    }

    public void start() {
        int choice;

        System.out.println("================================");
        System.out.println("       WELCOME TO JAVA ATM");
        System.out.println("================================");

        do {
            System.out.println("\n------ ATM MENU ------");
            System.out.println("1. Check Balance");
            System.out.println("2. Deposit Money");
            System.out.println("3. Withdraw Money");
            System.out.println("4. Exit");
            System.out.print("Enter your choice: ");

            choice = readInteger();

            switch (choice) {
                case 1:
                    account.checkBalance();
                    break;

                case 2:
                    System.out.print("Enter amount to deposit: Rs. ");
                    account.deposit(readAmount());
                    break;

                case 3:
                    System.out.print("Enter amount to withdraw: Rs. ");
                    account.withdraw(readAmount());
                    break;

                case 4:
                    System.out.println("Thank you for using Java ATM!");
                    System.out.println("Have a nice day!");
                    break;

                default:
                    System.out.println(
                        "Invalid choice! Please select 1 to 4."
                    );
            }
        } while (choice != 4);
    }

    // Validate menu input
    private int readInteger() {
        while (!scanner.hasNextInt()) {
            System.out.print("Invalid input! Enter a number: ");
            scanner.next();
        }
        return scanner.nextInt();
    }

    // Validate monetary input
    private double readAmount() {
        while (!scanner.hasNextDouble()) {
            System.out.print("Invalid input! Enter a valid amount: Rs. ");
            scanner.next();
        }
        return scanner.nextDouble();
    }
}

// Main class
public class ATMInterface {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Create a bank account with Rs. 5000
        BankAccount account = new BankAccount(5000.00);

        // Create ATM object
        ATM atm = new ATM(account, scanner);

        // Start the ATM
        atm.start();

        scanner.close();
    }
